### 实现要求(对 s05 代码的精确扩展)

#### A. RealtimeHub(纯 Python pub/sub)
```python
class RealtimeHub:
    """task/issue → 订阅者集合。纯逻辑,不依赖 WS。"""

    def __init__(self):
        self._subs: dict[str, set] = {}   # id -> set of asyncio.Queue or callable
        self._lock = threading.Lock()

    def subscribe(self, key: str) -> "queue.Queue":
        """订阅某 key(task_id 或 issue_id),返回一个 Queue 接收事件。"""
        import queue
        q = queue.Queue()
        with self._lock:
            self._subs.setdefault(key, set()).add(q)
        return q

    def unsubscribe(self, key: str, q):
        with self._lock:
            if key in self._subs:
                self._subs[key].discard(q)
                if not self._subs[key]:
                    del self._subs[key]

    def publish(self, key: str, event: dict):
        """推事件给 key 的所有订阅者。"""
        with self._lock:
            subs = list(self._subs.get(key, set()))
        for q in subs:
            q.put(event)   # 非阻塞 put(Queue 默认无界)

    def subscriber_count(self, key: str) -> int:
        with self._lock:
            return len(self._subs.get(key, set()))
```

- `_publish(self, task_id, event)`:若 `self.hub`:`self.hub.publish(task_id, event)`。
- `execute` 里 `on_event` 调 `self._publish(task.id, ev)`(在 `_record` 旁边)。

#### C. WS server(asyncio + websockets,薄胶水)
```python
import asyncio
import websockets

async def _ws_handler(websocket, path, hub):
    """从 path 解析 task_id,订阅 hub,推事件给 WS 客户端。"""
    # path 格式:/watch/<task_id>
    parts = path.strip("/").split("/")
    if len(parts) < 2 or parts[0] != "watch":
        await websocket.close(code=4000, reason="path must be /watch/<task_id>")
        return
    task_id = parts[1]
    q = hub.subscribe(task_id)
    try:
        while True:
            event = await asyncio.get_event_loop().run_in_executor(None, q.get, True)
            if event is None:  # 哨兵,退出
                break
            await websocket.send(json.dumps(event, ensure_ascii=False))
    except Exception:
        pass
    finally:
        hub.unsubscribe(task_id, q)

async def _ws_server(hub, port):
    async with websockets.serve(lambda ws, path: _ws_handler(ws, path, hub), "0.0.0.0", port):
        await asyncio.Future()  # 永久运行

def run_ws_server(hub, port):
    """同步入口:启动 WS server(阻塞)。"""
    asyncio.run(_ws_server(hub, port))
```

#### D. 终端客户端 watch_task
```python
def watch_task(host: str, port: int, task_id: str):
    """连 WS 订阅 task,实时打印事件。"""
    import websockets
    url = f"ws://{host}:{port}/watch/{task_id}"
    print(f"watching {task_id} at {url} ...")
    async def _watch():
        async with websockets.connect(url) as ws:
            while True:
                msg = await ws.recv()
                print(msg)
    asyncio.run(_watch())
```

### 测试 `tests/test_realtime.py`
```python
from sessions.zh.s06_realtime import RealtimeHub


def test_subscribe_and_publish():
    hub = RealtimeHub()
    q = hub.subscribe("task-1")
    hub.publish("task-1", {"type": "progress", "message": "hi"})
    ev = q.get(timeout=1)
    assert ev["message"] == "hi"


def test_multiple_subscribers_broadcast():
    hub = RealtimeHub()
    q1 = hub.subscribe("task-1")
    q2 = hub.subscribe("task-1")
    hub.publish("task-1", {"type": "done"})
    assert q1.get(timeout=1)["type"] == "done"
    assert q2.get(timeout=1)["type"] == "done"


def test_late_subscriber_no_replay():
    """晚到订阅者不收历史事件(只收订阅之后的新事件)。"""
    hub = RealtimeHub()
    hub.publish("task-1", {"type": "progress", "msg": "old"})  # 先发(无人订阅)
    q = hub.subscribe("task-1")  # 后订阅
    hub.publish("task-1", {"type": "progress", "msg": "new"})
    ev = q.get(timeout=1)
    assert ev["msg"] == "new"  # 只收到订阅之后的
    assert q.empty()           # 没有历史回放


def test_unsubscribe_stops_receiving():
    hub = RealtimeHub()
    q = hub.subscribe("task-1")
    hub.unsubscribe("task-1", q)
    hub.publish("task-1", {"type": "progress"})
    assert q.empty()  # 退订后不收


def test_different_keys_isolated():
    hub = RealtimeHub()
    q1 = hub.subscribe("task-1")
    q2 = hub.subscribe("task-2")
    hub.publish("task-1", {"type": "done"})
    assert not q1.empty()
    assert q2.empty()  # task-2 不收 task-1 的事件


def test_subscriber_count():
    hub = RealtimeHub()
    assert hub.subscriber_count("task-1") == 0
    hub.subscribe("task-1")
    hub.subscribe("task-1")
    assert hub.subscriber_count("task-1") == 2
```
