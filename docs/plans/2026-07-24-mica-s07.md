#### A. Daemon._publish 额外按 issue_id 推
s06 的 `_publish(task_id, event)` 只推 task_id。s07 改为额外推 issue_id:
```python
def _publish(self, task_id, event, issue_id=None):
    if self.hub:
        self.hub.publish(task_id, event)
        if issue_id:
            self.hub.publish(issue_id, event)
```
Daemon.execute 里调 `self._publish(task.id, ev, issue_id=task.issue_id)`。

#### B. create_app(issue_store, task_store, hub) → FastAPI app
```python
from fastapi import FastAPI, WebSocket
from fastapi.responses import HTMLResponse

def create_app(issue_store, task_store, hub):
    app = FastAPI()

    @app.get("/", response_class=HTMLResponse)
    def index():
        return HTML_PAGE  # 内联 HTML+JS

    @app.get("/api/issues")
    def list_issues():
        return [_issue_to_dict(i) for i in issue_store.list()]

    @app.post("/api/issues")
    def create_issue(title: str = "", assignee_type: str = "agent", assignee_id: str = ""):
        issue = issue_store.create("ws-1", title or "(untitled)",
                                    assignee_type=assignee_type, assignee_id=assignee_id)
        if assignee_type == "agent" and assignee_id:
            task_store.enqueue(issue.id, assignee_type, assignee_id)
        return _issue_to_dict(issue)

    @app.patch("/api/issues/{issue_id}/status")
    def update_status(issue_id: str, status: str):
        try:
            issue_store.transition(issue_id, IssueStatus(status))
        except (KeyError, IllegalTransition, ValueError) as e:
            return {"error": str(e)}, 400
        return _issue_to_dict(issue_store.get(issue_id))

    @app.post("/api/issues/{issue_id}/comments")
    def add_comment(issue_id: str, text: str = ""):
        issue = issue_store.comment(issue_id, text)
        task = task_store.enqueue(issue_id, issue.assignee_type, issue.assignee_id,
                                  trigger_comment_id=text)
        return {"issue": _issue_to_dict(issue), "task_id": task.id}

    @app.websocket("/ws/issues/{issue_id}")
    async def ws_issues(websocket: WebSocket, issue_id: str):
        await websocket.accept()
        q = hub.subscribe(issue_id)
        try:
            while True:
                event = q.get(timeout=0.5)
                if event:
                    await websocket.send_json(event)
        except Exception:
            pass
        finally:
            hub.unsubscribe(issue_id, q)

    return app
```

#### C. HTML_PAGE(内联 HTML+JS,零构建)
一个简洁的单页:issue 行表 + 创建表单(title + assignee) + 行内状态下拉(in_progress→in_review→done 等) + 评论框。JS 用 `fetch` 调 API,用 `WebSocket` 连 `/ws/issues/{id}` 实时更新选中行。

```python
HTML_PAGE = """<!DOCTYPE html>
<html><head><meta charset="utf-8"><title>mica board</title></head>
<body>
<h1>mica board</h1>
<form id="createForm">
  <input name="title" placeholder="title" required>
  <input name="assignee_id" placeholder="agent id" value="agent-1">
  <button type="submit">Create</button>
</form>
<table id="board" border="1">
  <tr><th>ID</th><th>Status</th><th>Assignee</th><th>Title</th><th>Action</th></tr>
</table>
<div id="progress"></div>
<script>
async function refresh() {
  const res = await fetch('/api/issues');
  const issues = await res.json();
  const tbody = document.querySelector('#board');
  tbody.innerHTML = '<tr><th>ID</th><th>Status</th><th>Assignee</th><th>Title</th><th>Action</th></tr>';
  issues.forEach(i => {
    const tr = document.createElement('tr');
    tr.innerHTML = `<td>${i.id.slice(0,8)}</td><td>${i.status}</td>
      <td>${i.assignee_type}:${i.assignee_id||'-'}</td><td>${i.title}</td>
      <td><button onclick="watch('${i.id}')">watch</button>
      <select onchange="transition('${i.id}', this.value)">
        <option value="">change status...</option>
        <option value="in_progress">in_progress</option>
        <option value="in_review">in_review</option>
        <option value="done">done</option>
        <option value="blocked">blocked</option>
      </select></td>`;
    tbody.appendChild(tr);
  });
}
document.getElementById('createForm').onsubmit = async (e) => {
  e.preventDefault();
  const f = new FormData(e.target);
  await fetch('/api/issues?title=' + encodeURIComponent(f.get('title'))
    + '&assignee_type=agent&assignee_id=' + encodeURIComponent(f.get('assignee_id')),
    {method:'POST'});
  refresh();
};
async function transition(id, status) {
  if (!status) return;
  await fetch('/api/issues/' + id + '/status?status=' + status, {method:'PATCH'});
  refresh();
}
function watch(id) {
  const ws = new WebSocket('ws://' + location.host + '/ws/issues/' + id);
  ws.onmessage = (e) => {
    document.getElementById('progress').textContent += e.data + '\\n';
  };
}
refresh();
</script>
</body></html>
"""
```

#### D. main(demo:启动 FastAPI + daemon 后台线程)
```python
def main(data_dir="./workspace"):
    import threading, uvicorn
    os.makedirs(data_dir, exist_ok=True)
    issue_store = IssueStore(data_dir=data_dir)
    task_store = TaskStore(data_dir=data_dir)
    sessions = SessionStore(data_dir)
    hub = RealtimeHub()
    agent = Agent(id="agent-1", name="Claude")
    daemon = Daemon(issue_store, task_store, agent, StubRunner(),
                    session_store=sessions, hub=hub)
    # daemon 后台轮询
    t = threading.Thread(target=daemon.loop, daemon=True)
    t.start()
    app = create_app(issue_store, task_store, hub)
    print("--- mica board on http://localhost:8000 ---")
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

### 测试 `tests/test_web_panel.py`
```python
import pytest
from fastapi.testclient import TestClient
from sessions.zh.s07_web_panel import (
    create_app, IssueStore, TaskStore, RealtimeHub, IssueStatus, IllegalTransition,
    _issue_to_dict,
)


def _client():
    is_ = IssueStore()
    ts = TaskStore()
    hub = RealtimeHub()
    app = create_app(is_, ts, hub)
    return TestClient(app), is_, ts


def test_list_issues_empty():
    client, _, _ = _client()
    res = client.get("/api/issues")
    assert res.json() == []


def test_create_issue():
    client, is_, ts = _client()
    res = client.post("/api/issues?title=test&assignee_type=agent&assignee_id=a1")
    data = res.json()
    assert data["title"] == "test"
    assert data["status"] == "todo"
    assert data["assignee_id"] == "a1"
    # task 入队
    assert len(ts.list(data["id"])) == 1


def test_status_transition_legal():
    client, is_, _ = _client()
    issue = is_.create("ws-1", "t")
    is_.transition(issue.id, IssueStatus.IN_PROGRESS)
    res = client.patch(f"/api/issues/{issue.id}/status?status=in_review")
    assert res.json()["status"] == "in_review"


def test_status_transition_illegal_rejected():
    client, is_, _ = _client()
    issue = is_.create("ws-1", "t")
    res = client.patch(f"/api/issues/{issue.id}/status?status=done")
    # todo→done is illegal (must go through in_progress)
    assert res.status_code == 200  # FastAPI returns 200 with error body
    assert "error" in res.json()


def test_comment_creates_new_task():
    client, is_, ts = _client()
    issue = is_.create("ws-1", "t", assignee_type="agent", assignee_id="a1")
    res = client.post(f"/api/issues/{issue.id}/comments?text=fix+this")
    data = res.json()
    assert data["issue"]["comments"] == ["fix this"]
    assert data["task_id"]  # task 入队
    tasks = ts.list(issue.id)
    assert len(tasks) == 1
    assert tasks[0].trigger_comment_id == "fix this"


def test_get_html_page():
    client, _, _ = _client()
    res = client.get("/")
    assert res.status_code == 200
    assert "<html" in res.text.lower()
    assert "mica board" in res.text
```
