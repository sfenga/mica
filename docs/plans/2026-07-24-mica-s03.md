#### A. parse_stream_line(纯函数 — 与旧 s03 同,但 result 终态带 session_id)
```python
def parse_stream_line(line: str) -> tuple[list[dict], dict | None]:
    s = line.strip()
    if not s:
        return [], None
    try:
        evt = json.loads(s)
    except json.JSONDecodeError:
        return [], None
    events: list[dict] = []
    terminal: dict | None = None
    t = evt.get("type")
    if t == "system":
        if evt.get("subtype") in ("thinking_tokens", "hook_started", "hook_response"):
            pass
        else:
            events.append({"kind": "system", "session_id": evt.get("session_id", "")})
    elif t == "assistant":
        for block in evt.get("message", {}).get("content", []):
            bt = block.get("type")
            if bt == "text":
                events.append({"kind": "text", "text": block.get("text", "")})
            elif bt == "thinking":
                events.append({"kind": "thinking", "text": block.get("text", "")})
            elif bt == "tool_use":
                events.append({"kind": "tool_use", "tool": block.get("name", "")})
    elif t == "user":
        for block in evt.get("message", {}).get("content", []):
            if block.get("type") == "tool_result":
                events.append({"kind": "tool_result",
                               "tool_use_id": block.get("tool_use_id", ""),
                               "is_error": bool(block.get("is_error", False))})
    elif t == "result":
        is_error = bool(evt.get("is_error")) or evt.get("subtype") == "error"
        terminal = {"status": "failed" if is_error else "done",
                    "output": evt.get("result", ""), "is_error": is_error,
                    "session_id": evt.get("session_id", "")}
    elif t == "log":
        events.append({"kind": "log", "message": evt.get("log", {}).get("message", "")})
    return events, terminal
```

#### B. ClaudeCliRunner(适配新签名 run(task, issue, ...))
```python
class ClaudeCliRunner:
    """spawn `claude -p --verbose --output-format stream-json`,逐行解析、转发、fail-closed。"""

    def __init__(self, cli: str = "claude") -> None:
        self.cli = cli

    def _cmd(self, prompt: str, append_system_prompt: str | None = None,
             resume_session_id: str | None = None) -> list[str]:
        cmd = [self.cli, "-p", "--verbose", "--output-format", "stream-json", prompt]
        if append_system_prompt:
            cmd += ["--append-system-prompt", append_system_prompt]
        if resume_session_id:
            cmd += ["--resume", resume_session_id]
        return cmd   # SHALL NOT 含 --no-session-persistence

    def run(self, task, issue, context: list, on_event,
            resume_session_id: str | None = None) -> dict:
        prompt = issue.title + (f"\n\n{issue.body}" if issue.body else "")
        try:
            proc = subprocess.Popen(
                self._cmd(prompt, resume_session_id=resume_session_id),
                stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True,
            )
        except FileNotFoundError:
            return {"status": "failed", "output": f"CLI not found: {self.cli}",
                    "exit_code": 127, "session_id": ""}
        terminal: dict | None = None
        captured_sid = ""
        for line in proc.stdout:
            events, term = parse_stream_line(line)
            for ev in events:
                if ev.get("kind") == "system" and ev.get("session_id"):
                    captured_sid = ev["session_id"]
                ev["task_id"] = task.id
                on_event(ev)
            if term is not None:
                terminal = term
                if term.get("session_id"):
                    captured_sid = term["session_id"]
        proc.stdout.close()
        stderr = proc.stderr.read() if proc.stderr else ""
        rc = proc.wait()
        if terminal is None:
            return {"status": "failed",
                    "output": stderr or f"stream ended without result event (exit {rc})",
                    "exit_code": rc, "session_id": captured_sid}
        if terminal["is_error"]:
            return {"status": "failed",
                    "output": terminal["output"] or stderr or "result is_error",
                    "exit_code": rc, "session_id": captured_sid}
        if rc != 0:
            return {"status": "failed", "output": stderr or f"exit {rc}",
                    "exit_code": rc, "session_id": captured_sid}
        return {"status": "done", "output": terminal["output"],
                "exit_code": rc, "session_id": captured_sid}
```

import json
from sessions.zh.s03_claude_runner import parse_stream_line


def line(d):

def test_system_thinking_tokens_dropped():
    events, term = parse_stream_line(line({"type": "system", "subtype": "thinking_tokens",

    assert events == [] and term is None


def test_assistant_text_and_tool_use_forwarded():
    ev_line = {"type": "assistant", "message": {"content": [

`tests/test_cli_runner_finalize.py`(monkeypatched Popen,验 finalize + session_id 捕获,不 spawn claude):
```python
import io
import sessions.zh.s03_claude_runner as m
from sessions.zh.s03_claude_runner import ClaudeCliRunner, Issue, Task


class _FakeProc:
    def __init__(self, stdout_lines, stderr="", returncode=0):
        self.stdout = io.StringIO("".join(stdout_lines))
        self.stderr = io.StringIO(stderr)
        self._rc = returncode
    def wait(self):
        return self._rc


def _patch(monkeypatch, proc, capture=None):
    def fake(cmd, **k):
        if capture is not None:
            capture.append(cmd)
        return proc
    monkeypatch.setattr(m.subprocess, "Popen", fake)


def _issue():
    return Issue(id="i1", workspace_id="ws", title="the-original-task")


def _task():
    return Task(id="t1", issue_id="i1", assignee_type="agent", assignee_id="a1")


def test_no_result_event_marks_failed(monkeypatch):
    _patch(monkeypatch, _FakeProc([
        '{"type":"system","subtype":"init","session_id":"ses-1"}\n',
        '{"type":"assistant","message":{"content":[{"type":"text","text":"hi"}]}}\n',
    ]))
    res = ClaudeCliRunner().run(_task(), _issue(), [], lambda e: None)
    assert res["status"] == "failed"
    assert res["session_id"] == "ses-1"   # system init 的 session_id 仍捕获
    assert "without result" in res["output"]


def test_result_error_marks_failed(monkeypatch):
    _patch(monkeypatch, _FakeProc([
        '{"type":"result","subtype":"error","result":"boom","is_error":true,"session_id":"ses-2"}\n',
    ]))
    res = ClaudeCliRunner().run(_task(), _issue(), [], lambda e: None)
    assert res["status"] == "failed" and res["output"] == "boom" and res["session_id"] == "ses-2"


def test_result_success_exit_zero_done(monkeypatch):
    captured = []
    _patch(monkeypatch, _FakeProc([
        '{"type":"assistant","message":{"content":[{"type":"text","text":"hi"}]}}\n',
        '{"type":"result","subtype":"success","result":"hi","is_error":false,"session_id":"ses-3"}\n',
    ]))
    res = ClaudeCliRunner().run(_task(), _issue(), [], captured.append)
    assert res["status"] == "done" and res["output"] == "hi" and res["session_id"] == "ses-3"
    assert any(e.get("kind") == "text" for e in captured)


def test_result_success_but_exit_nonzero_failed(monkeypatch):
    _patch(monkeypatch, _FakeProc([
        '{"type":"result","subtype":"success","result":"ok","is_error":false,"session_id":"ses-4"}\n',
    ], rc=2))
    res = ClaudeCliRunner().run(_task(), _issue(), [], lambda e: None)
    assert res["status"] == "failed" and res["session_id"] == "ses-4"


def test_cli_not_found(monkeypatch):
    def boom(*a, **k):
        raise FileNotFoundError(2, "no")
    monkeypatch.setattr(m.subprocess, "Popen", boom)
    res = ClaudeCliRunner(cli="nonexistent-cli-xyz").run(_task(), _issue(), [], lambda e: None)
    assert res["status"] == "failed" and "not found" in res["output"] and res["session_id"] == ""


def test_no_resume_when_none(monkeypatch):
    argv = []
    _patch(monkeypatch, _FakeProc([
        '{"type":"result","subtype":"success","result":"ok","is_error":false,"session_id":"ses-5"}\n',
    ]), capture=argv)
    ClaudeCliRunner().run(_task(), _issue(), [], lambda e: None)
    assert "--resume" not in argv[0]
    assert "--no-session-persistence" not in argv[0]
```
