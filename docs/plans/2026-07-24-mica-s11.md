#### B. WorkspaceScope(过滤 + 成员校验)
```python
class WorkspaceScope:
    """在 IssueStore/TaskStore 查询时注入 workspace_id 过滤 + 成员校验。"""
    def __init__(self, workspace_id: str, member_id: str, member_store: MemberStore):
        self.workspace_id = workspace_id
        self.member_id = member_id
        self.member_store = member_store

    def check_access(self):
        """校验当前 member 属于该 workspace;非成员抛 PermissionError。"""
        if not self.member_store.is_member(self.workspace_id, self.member_id):
            raise PermissionError(
                f"member {self.member_id} not in workspace {self.workspace_id}")
```

#### D. main(demo:两个 workspace + 成员校验)
```python
def main(data_dir="./workspace"):
    os.makedirs(data_dir, exist_ok=True)
    ms = MemberStore(data_dir=data_dir)
    ms.add_member("ws-a", "alice")
    ms.add_member("ws-b", "bob")
    issue_store = IssueStore(data_dir=data_dir)
    issue_store.create("ws-a", "task A", assignee_type="agent", assignee_id="agent-1")
    issue_store.create("ws-b", "task B", assignee_type="agent", assignee_id="agent-1")
    # alice 看 ws-a(alice 是成员)
    print("alice sees ws-a:", [i.title for i in issue_store.list("ws-a", "alice", ms)])
    # alice 看 ws-b(非成员 → PermissionError)
    try:
        issue_store.list("ws-b", "alice", ms)
    except PermissionError as e:
        print(f"alice denied ws-b: {e}")
    return 0
```
