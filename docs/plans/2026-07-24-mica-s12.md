#### B. SquadRouter
```python
class SquadRouter:
    """leader agent 把派给 squad 的 issue 委派给成员 agent。"""

    def __init__(self, issue_store, task_store):
        self.issue_store = issue_store
        self.task_store = task_store
        self._squads: dict[str, Squad] = {}

    def register(self, squad: Squad):
        self._squads[squad.id] = squad

    def route(self, issue_id, squad_id) -> str | None:
        """派给 squad 的 issue:leader 选成员 agent → 创建子 task。
        返回成员 agent_id,或 None(squad 无成员)。"""
        squad = self._squads.get(squad_id)
        if not squad or not squad.member_agent_ids:
            return None
        # 简化:轮流选(round-robin)
        import hashlib
        idx = int(hashlib.md5(issue_id.encode()).hexdigest(), 16) % len(squad.member_agent_ids)
        chosen = squad.member_agent_ids[idx]
        # 为成员 agent 入队 task(issue 不变,assignee 换成成员 agent)
        self.task_store.enqueue(issue_id, "agent", chosen)
        return chosen

    def list_squads(self) -> list[Squad]:
        return list(self._squads.values())
```

#### C. main(demo)
```python
def main(data_dir="./workspace"):
    os.makedirs(data_dir, exist_ok=True)
    is_ = IssueStore(data_dir=data_dir)
    ts = TaskStore(data_dir=data_dir)
    router = SquadRouter(is_, ts)
    squad = Squad(id="sq-1", name="Frontend Team",
                  leader_agent_id="agent-1",
                  member_agent_ids=["agent-2", "agent-3"])
    router.register(squad)
    issue = is_.create("ws-1", "build landing page",
                       assignee_type="agent", assignee_id="agent-1")
    chosen = router.route(issue.id, "sq-1")
    print(f"squad routed to: {chosen}")
    tasks = ts.list(issue.id)
    print(f"tasks: {len(tasks)}, assignee: {tasks[0].assignee_id if tasks else 'none'}")
    return 0
```

### 测试 `tests/test_squads.py`
```python
from sessions.zh.s12_squads import Squad, SquadRouter, IssueStore, TaskStore


def test_squad_registered():
    router = SquadRouter(IssueStore(), TaskStore())
    squad = Squad(id="sq-1", name="team", leader_agent_id="a1",
                  member_agent_ids=["a2", "a3"])
    router.register(squad)
    assert len(router.list_squads()) == 1
    assert router.list_squads()[0].name == "team"


def test_route_creates_task_for_member():
    is_ = IssueStore()
    ts = TaskStore()
    router = SquadRouter(is_, ts)
    squad = Squad(id="sq-1", name="t", leader_agent_id="a1",
                  member_agent_ids=["a2", "a3"])
    router.register(squad)
    issue = is_.create("ws-1", "build", assignee_type="agent", assignee_id="a1")
    chosen = router.route(issue.id, "sq-1")
    assert chosen in ["a2", "a3"]
    tasks = ts.list(issue.id)
    assert len(tasks) == 1
    assert tasks[0].assignee_id == chosen


def test_route_no_members_returns_none():
    is_ = IssueStore()
    ts = TaskStore()
    router = SquadRouter(is_, ts)
    squad = Squad(id="sq-1", name="empty", leader_agent_id="a1",
                  member_agent_ids=[])
    router.register(squad)
    issue = is_.create("ws-1", "t", assignee_type="agent", assignee_id="a1")
    assert router.route(issue.id, "sq-1") is None


def test_route_unknown_squad_returns_none():
    router = SquadRouter(IssueStore(), TaskStore())
    issue = IssueStore().create("ws-1", "t", assignee_type="agent", assignee_id="a1")
    assert router.route(issue.id, "nonexistent") is None
```
