```python
"""s01 — Issue 模型 + 看板 + 多态 assignee + 状态机

对应 multiCa:issue lifecycle / agents as teammates。
状态机:todo → in_progress → done/failed。failed 为终态。
多态 assignee:assignee_type ∈ {"", "member", "agent"} + assignee_id。
本章 store 为内存版;s04 起换 JSONL 后端。
"""
import sys
import uuid
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


class IssueStatus(str, Enum):
    TODO = "todo"
    IN_PROGRESS = "in_progress"
    DONE = "done"
    FAILED = "failed"


# 合法状态转移:键=当前状态,值=可转到的状态集合。其它转移一律拒绝。
VALID_TRANSITIONS = {
    IssueStatus.TODO: {IssueStatus.IN_PROGRESS},
    IssueStatus.IN_PROGRESS: {IssueStatus.DONE, IssueStatus.FAILED},
    IssueStatus.DONE: set(),
    IssueStatus.FAILED: set(),
}


@dataclass
class Workspace:
    id: str
    slug: str
    name: str


@dataclass
class Agent:
    id: str
    name: str
    cli: str = "claude"
    runtime_id: str = "local"


@dataclass
class Member:
    id: str
    name: str


@dataclass
class Issue:
    id: str
    workspace_id: str
    title: str
    body: str = ""
    status: IssueStatus = IssueStatus.TODO
    assignee_type: str = ""       # "" | "member" | "agent"
    assignee_id: str = ""
    created_at: str = field(default_factory=now_iso)
    updated_at: str = field(default_factory=now_iso)
    error: str = ""


class IllegalTransition(Exception):
    pass


class IssueStore:
    """内存 issue 存储 + 状态机校验。s04 加 JSONL 后端。"""

    def __init__(self) -> None:
        self._issues: dict[str, Issue] = {}

    def create(self, workspace_id: str, title: str, body: str = "",
               assignee_type: str = "", assignee_id: str = "") -> Issue:
        issue = Issue(
            id=str(uuid.uuid4()),
            workspace_id=workspace_id,
            title=title,
            body=body,
            assignee_type=assignee_type,
            assignee_id=assignee_id,
        )
        self._issues[issue.id] = issue
        return issue

    def list(self, workspace_id: str | None = None) -> list[Issue]:
        return [
            i for i in self._issues.values()
            if workspace_id is None or i.workspace_id == workspace_id
        ]

    def get(self, issue_id: str) -> Issue | None:
        return self._issues.get(issue_id)

    def assign(self, issue_id: str, assignee_type: str, assignee_id: str) -> Issue:
        issue = self._require(issue_id)
        issue.assignee_type = assignee_type
        issue.assignee_id = assignee_id
        issue.updated_at = now_iso()
        return issue

    def transition(self, issue_id: str, new_status: IssueStatus) -> Issue:
        issue = self._require(issue_id)
        if new_status not in VALID_TRANSITIONS[issue.status]:
            raise IllegalTransition(
                f"非法转移:{issue.status.value} -> {new_status.value}"
            )
        issue.status = new_status
        issue.updated_at = now_iso()
        return issue

    def _require(self, issue_id: str) -> Issue:
        issue = self._issues.get(issue_id)
        if issue is None:
            raise KeyError(f"issue 不存在:{issue_id}")
        return issue


def render_board(issues: list[Issue]) -> str:
    """终端看板:一行一个 issue。"""
    if not issues:
        return "(空看板)"
    lines = ["ID            STATUS       ASSIGNEE                TITLE"]
    for i in issues:
        assignee = f"{i.assignee_type}:{i.assignee_id}" if i.assignee_type else "-"
        lines.append(f"{i.id[:12]:<13} {i.status.value:<12} {assignee:<23} {i.title}")
    return "\n".join(lines)


# --- CLI(单进程内执行;s04 持久化后可跨进程)---
def cmd_main(argv: list[str]) -> int:
    store = IssueStore()
    ws_id = "ws-1"
    if not argv:
        # 无参:演示场景
        a = store.create(ws_id, "修登录 bug", assignee_type="agent", assignee_id="agent-1")
        store.create(ws_id, "写文档", assignee_type="member", assignee_id="alice")
        store.transition(a.id, IssueStatus.IN_PROGRESS)
        print(render_board(store.list()))
        return 0
    cmd = argv[0]
    if cmd == "create":
        title = argv[1] if len(argv) > 1 else "(untitled)"
        at = argv[2] if len(argv) > 2 else ""
        aid = argv[3] if len(argv) > 3 else ""
        issue = store.create("ws-1", title, assignee_type=at, assignee_id=aid)
        print(f"created {issue.id} '{issue.title}'")
        return 0
    if cmd == "list":
        print(render_board(store.list()))
        return 0
    if cmd == "assign":
        store.assign(argv[1], argv[2], argv[3])
        print("assigned")
        return 0
    print(f"未知命令:{cmd}", file=sys.stderr)
    return 2


if __name__ == "__main__":
    sys.exit(cmd_main(sys.argv[1:]))
```

"""s02 — Daemon 认领执行循环(stub runner)

对应 multiCa:daemon runtime。
前置:s01 的 Issue 模型与状态机(原样保留),本章给 IssueStore 加带锁的 claim()/set_error(),
再加 AgentRunner 抽象 + StubRunner + Daemon。
s03 起 StubRunner 让位给 ClaudeCliRunner。
"""
import sys
import threading
import time
import uuid
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from typing import Callable, Protocol, TypedDict


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


class IssueStatus(str, Enum):
    TODO = "todo"
    IN_PROGRESS = "in_progress"
    DONE = "done"
    FAILED = "failed"


VALID_TRANSITIONS = {
    IssueStatus.TODO: {IssueStatus.IN_PROGRESS},
    IssueStatus.IN_PROGRESS: {IssueStatus.DONE, IssueStatus.FAILED},
    IssueStatus.DONE: set(),
    IssueStatus.FAILED: set(),
}


@dataclass
class Agent:
    id: str
    name: str
    cli: str = "claude"
    runtime_id: str = "local"


@dataclass
class Issue:
    id: str
    workspace_id: str
    title: str
    body: str = ""
    status: IssueStatus = IssueStatus.TODO
    assignee_type: str = ""
    assignee_id: str = ""
    created_at: str = field(default_factory=now_iso)
    updated_at: str = field(default_factory=now_iso)
    error: str = ""


class IllegalTransition(Exception):
    pass


class IssueStore:
    """内存 issue 存储 + 状态机 + CAS 认领。"""

    def __init__(self) -> None:
        self._issues: dict[str, Issue] = {}
        self._lock = threading.Lock()

    def create(self, workspace_id: str, title: str, body: str = "",
               assignee_type: str = "", assignee_id: str = "") -> Issue:
        with self._lock:
            issue = Issue(
                id=str(uuid.uuid4()),
                workspace_id=workspace_id,
                title=title,
                body=body,
                assignee_type=assignee_type,
                assignee_id=assignee_id,
            )
            self._issues[issue.id] = issue
            return issue

    def list(self, workspace_id: str | None = None) -> list[Issue]:
        with self._lock:
            return [
                i for i in self._issues.values()
                if workspace_id is None or i.workspace_id == workspace_id
            ]

    def get(self, issue_id: str) -> Issue | None:
        with self._lock:
            return self._issues.get(issue_id)

    def assign(self, issue_id: str, assignee_type: str, assignee_id: str) -> Issue:
        with self._lock:
            issue = self._require(issue_id)
            issue.assignee_type = assignee_type
            issue.assignee_id = assignee_id
            issue.updated_at = now_iso()
            return issue

    def transition(self, issue_id: str, new_status: IssueStatus) -> Issue:
        with self._lock:
            issue = self._require(issue_id)
            if new_status not in VALID_TRANSITIONS[issue.status]:
                raise IllegalTransition(
                    f"非法转移:{issue.status.value} -> {new_status.value}"
                )
            issue.status = new_status
            issue.updated_at = now_iso()
            return issue

    def claim(self, issue_id: str) -> bool:
        """CAS 认领:仅当当前 status==todo 才置 in_progress。返回是否成功。"""
        with self._lock:
            issue = self._issues.get(issue_id)
            if issue is None or issue.status != IssueStatus.TODO:
                return False
            issue.status = IssueStatus.IN_PROGRESS
            issue.updated_at = now_iso()
            return True

    def set_error(self, issue_id: str, error: str) -> None:
        with self._lock:
            issue = self._require(issue_id)
            issue.error = error
            issue.updated_at = now_iso()

    def _require(self, issue_id: str) -> Issue:
        issue = self._issues.get(issue_id)
        if issue is None:
            raise KeyError(f"issue 不存在:{issue_id}")
        return issue


# --- AgentRunner 抽象 ---
class RunResult(TypedDict):
    status: str          # "done" | "failed"
    output: str
    exit_code: int


class AgentRunner(Protocol):
    def run(self, issue: Issue, context: list,
            on_event: Callable[[dict], None]) -> RunResult: ...


class StubRunner:
    """s02 占位 runner:发合成事件、无子进程、不调模型。s03 换 ClaudeCliRunner。"""

    def run(self, issue: Issue, context: list,
            on_event: Callable[[dict], None]) -> RunResult:
        on_event({"type": "progress", "issue_id": issue.id,
                  "message": f"stub: start '{issue.title}'"})
        on_event({"type": "progress", "issue_id": issue.id,
                  "message": "stub: working..."})
        on_event({"type": "done", "issue_id": issue.id,
                  "output": f"stub completed: {issue.title}"})
        return {"status": "done", "output": f"stub completed: {issue.title}",
                "exit_code": 0}


# --- Daemon ---
class Daemon:
    def __init__(self, store: IssueStore, agent: Agent, runner: AgentRunner,
                 poll_interval: float = 2.0) -> None:
        self.store = store
        self.agent = agent
        self.runner = runner
        self.poll_interval = poll_interval
        self._stop = False

    def claim_next(self) -> Issue | None:
        """原子认领一条派给本 agent 的 todo issue;无则返回 None。"""
        candidates = [
            i for i in self.store.list()
            if i.assignee_type == "agent"
            and i.assignee_id == self.agent.id
            and i.status == IssueStatus.TODO
        ]
        for issue in candidates:
            if self.store.claim(issue.id):   # CAS:只有一个会成功
                return issue
        return None

    def execute(self, issue: Issue) -> RunResult:
        def on_event(ev: dict) -> None:
            print(f"[event] {ev}")
        result = self.runner.run(issue, [], on_event)
        if result["status"] == "done":
            self.store.transition(issue.id, IssueStatus.DONE)
        else:
            self.store.set_error(issue.id, result.get("output", ""))
            self.store.transition(issue.id, IssueStatus.FAILED)
        return result

    def tick(self) -> bool:
        issue = self.claim_next()
        if issue is None:
            return False
        print(f"[daemon] claimed {issue.id} '{issue.title}'")
        self.execute(issue)
        return True

    def run_once(self) -> None:
        """处理所有当前可认领的 issue 一次(测试用)。"""
        while self.tick():
            pass

    def loop(self) -> None:
        while not self._stop:
            self.tick()
            time.sleep(self.poll_interval)


def render_board(issues: list[Issue]) -> str:
    if not issues:
        return "(空看板)"
    lines = ["ID            STATUS       ASSIGNEE                TITLE"]
    for i in issues:
        assignee = f"{i.assignee_type}:{i.assignee_id}" if i.assignee_type else "-"
        lines.append(f"{i.id[:12]:<13} {i.status.value:<12} {assignee:<23} {i.title}")
    return "\n".join(lines)


def main() -> int:
    store = IssueStore()
    agent = Agent(id="agent-1", name="Claude")
    daemon = Daemon(store, agent, StubRunner())
    store.create("ws-1", "修登录 bug", assignee_type="agent", assignee_id="agent-1")
    store.create("ws-1", "写文档(给人)", assignee_type="member", assignee_id="alice")
    print("--- 派活前 ---")
    print(render_board(store.list()))
    daemon.run_once()
    print("--- 认领执行后 ---")
    print(render_board(store.list()))
    return 0


if __name__ == "__main__":
    sys.exit(main())
