#### C. Scheduler
```python
from croniter import croniter

class Scheduler:
    """cron 调度器:tick 检查到期 → 建 issue+task → 推进 next_run。"""

    def __init__(self, autopilot_store, issue_store, task_store, agent_id="agent-1"):
        self.ap_store = autopilot_store
        self.issue_store = issue_store
        self.task_store = task_store
        self.agent_id = agent_id

    def tick(self, now=None) -> int:
        """检查所有 enabled 的 autopilot,到期的触发。返回触发数。"""
        import datetime
        if now is None:
            now = datetime.datetime.now(datetime.timezone.utc)
        triggered = 0
        for ap in self.ap_store.list():
            if not ap.enabled:
                continue
            if not ap.next_run:
                # 初始化 next_run
                self._compute_next_run(ap, now)
                continue
            next_run = self._parse_iso(ap.next_run)
            if next_run <= now:
                self._fire(ap, now)
                triggered += 1
        return triggered

    def _fire(self, ap, now):
        """到期触发:建 issue+task,推进 next_run,更新 fail_count。"""
        # 建 issue
        issue = self.issue_store.create("ws-1", ap.title, body=ap.template,
                                        assignee_type="agent", assignee_id=ap.agent_id)
        # 入队 task
        self.task_store.enqueue(issue.id, "agent", ap.agent_id)
        # 推进 next_run(无论"执行"成功失败,触发即推进)
        self._compute_next_run(ap, now)
        # fail_count 清零(触发成功;实际"执行"结果由 daemon 异步返回,
        # 这里只记"触发了"——成功/失败由 daemon 执行后回写)
        # 简化:触发即清零(实际生产是执行成功才清零,但 mica 的 daemon 是异步的)
        ap.fail_count = 0
        self.ap_store.save(ap)

    def _compute_next_run(self, ap, now):
        """用 croniter 算 next_run。"""
        tz = self._get_tz(ap.timezone)
        local_now = now.astimezone(tz) if tz else now
        cron = croniter(ap.cron, local_now)
        next_dt = cron.get_next(datetime.datetime)
        ap.next_run = next_dt.isoformat()
        self.ap_store.save(ap)

    def _parse_iso(self, iso_str):
        import datetime
        return datetime.datetime.fromisoformat(iso_str)

    def _get_tz(self, tz_name):
        import datetime
        if tz_name == "UTC" or not tz_name:
            return datetime.timezone.utc
        try:
            import zoneinfo
            return zoneinfo.ZoneInfo(tz_name)
        except:
            return datetime.timezone.utc

    def mark_fail(self, ap_id):
        """daemon 执行失败时调:fail_count+1,连续 5 次禁用。"""
        ap = self.ap_store.get(ap_id)
        if not ap:
            return
        ap.fail_count += 1
        if ap.fail_count >= 5:
            ap.enabled = False
        self.ap_store.save(ap)

    def mark_success(self, ap_id):
        """daemon 执行成功时调:fail_count 清零。"""
        ap = self.ap_store.get(ap_id)
        if not ap:
            return
        ap.fail_count = 0
        self.ap_store.save(ap)
```

#### D. main(demo:创建 autopilot + tick 触发)
```python
def main(data_dir="./workspace"):
    os.makedirs(data_dir, exist_ok=True)
    ap_store = AutopilotStore(data_dir=data_dir)
    issue_store = IssueStore(data_dir=data_dir)
    task_store = TaskStore(data_dir=data_dir)
    scheduler = Scheduler(ap_store, issue_store, task_store)
    # 创建一条每分钟触发的 autopilot
    ap = Autopilot(id="ap-1", cron="* * * * *", agent_id="agent-1",
                   title="定时巡检", template="检查系统健康状态")
    scheduler._compute_next_run(ap, datetime.datetime.now(datetime.timezone.utc))
    ap_store.save(ap)
    print(f"autopilot: {ap.id} next_run={ap.next_run}")
    # 模拟到期(把 next_run 设到过去)
    ap.next_run = "2020-01-01T00:00:00+00:00"
    ap_store.save(ap)
    triggered = scheduler.tick()
    print(f"triggered: {triggered}")
    print(render_board(issue_store.list()))
    return 0
```

```python
import datetime, os, tempfile
from sessions.zh.s09_autopilot import Autopilot, AutopilotStore, Scheduler, IssueStore, TaskStore


def _now():
    return datetime.datetime.now(datetime.timezone.utc)


def test_cron_to_next_run():
    d = tempfile.mkdtemp()
    aps = AutopilotStore(data_dir=d)
    is_ = IssueStore()
    ts = TaskStore()
    sched = Scheduler(aps, is_, ts)
    ap = Autopilot(id="ap-1", cron="0 9 * * *", agent_id="a1", title="daily")
    sched._compute_next_run(ap, _now())
    assert ap.next_run  # ISO 时间字符串


def test_fire_creates_issue_and_task():
    d = tempfile.mkdtemp()
    aps = AutopilotStore(data_dir=d)
    is_ = IssueStore()
    ts = TaskStore()
    sched = Scheduler(aps, is_, ts)
    ap = Autopilot(id="ap-1", cron="* * * * *", agent_id="a1", title="tick")
    ap.next_run = "2020-01-01T00:00:00+00:00"  # 过去
    aps.save(ap)
    triggered = sched.tick(now=_now())
    assert triggered == 1
    issues = is_.list()
    assert len(issues) == 1
    assert issues[0].title == "tick"
    tasks = ts.list(issues[0].id)
    assert len(tasks) == 1


def test_fire_advances_next_run():
    d = tempfile.mkdtemp()
    aps = AutopilotStore(data_dir=d)
    sched = Scheduler(aps, IssueStore(), TaskStore())
    ap = Autopilot(id="ap-1", cron="0 9 * * *", agent_id="a1", title="daily")
    ap.next_run = "2020-01-01T00:00:00+00:00"
    aps.save(ap)
    sched.tick(now=_now())
    ap2 = aps.get("ap-1")

def test_consecutive_failures_disable():

    ap = Autopilot(id="ap-1", cron="* * * * *", agent_id="a1", title="t")

    sched = Scheduler(aps, IssueStore(), TaskStore())

        sched.mark_fail("ap-1")

def test_success_resets_fail_count():
    d = tempfile.mkdtemp()
    aps = AutopilotStore(data_dir=d)

    sched.mark_success("ap-1")
    assert aps.get("ap-1").fail_count == 0
```
