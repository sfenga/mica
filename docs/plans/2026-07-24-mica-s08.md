#### A. Skill dataclass
```python
@dataclass
class Skill:
    name: str
    description: str
    match: str          # 正则,作用于 issue.title + issue.body;空串=不自动选中
    body: str           # markdown 正文(skill 内容)
    path: str = ""      # SKILL.md 文件路径
```

#### B. SkillLoader
```python
import re, glob

class SkillLoader:
    """扫 workspace/skills/**/SKILL.md,按 frontmatter.match 正则选技能。"""

    def __init__(self, skills_dir: str):
        self._dir = skills_dir
        self._skills: list[Skill] = []
        self._load()

    def _load(self):
        """扫描目录,解析 frontmatter,加载技能列表。"""
        if not os.path.isdir(self._dir):
            return
        for path in sorted(glob.glob(os.path.join(self._dir, "**", "SKILL.md"), recursive=True)):
            skill = self._parse(path)
            if skill:
                self._skills.append(skill)

    def _parse(self, path: str) -> Skill | None:
        """解析 SKILL.md:---name/description/match--- + 正文。"""
        with open(path) as f:
            content = f.read()
        # 解析 frontmatter(--- 之间的 key: value)
        if not content.startswith("---"):
            return None
        parts = content.split("---", 2)
        if len(parts) < 3:
            return None
        fm = parts[1].strip()
        body = parts[2].strip()
        fields = {}
        for line in fm.splitlines():
            if ":" in line:
                k, v = line.split(":", 1)
                fields[k.strip()] = v.strip()
        return Skill(
            name=fields.get("name", ""),
            description=fields.get("description", ""),
            match=fields.get("match", ""),
            body=body,
            path=path,
        )

    def select(self, issue) -> Skill | None:
        """用 match 正则测 issue.title+body,取首个命中。无 match 字段的不选。"""
        text = issue.title + "\n" + (issue.body or "")
        for skill in self._skills:
            if skill.match and re.search(skill.match, text):
                return skill
        return None

    def list(self) -> list[Skill]:
        return list(self._skills)
```

import os, tempfile
from sessions.zh.s08_skills import Skill, SkillLoader, Issue, IssueStatus


def _tmp_skill(content):
    d = tempfile.mkdtemp()
    path = os.path.join(d, "SKILL.md")
    with open(path, "w") as f:
        f.write(content)
    return d, path


def test_match_selects_skill():
    d, _ = _tmp_skill("""---
name: code-review
description: review code
match: review|审.*查
---
Focus on security.""")
    loader = SkillLoader(d)
    issue = Issue(id="i1", workspace_id="ws", title="review the auth module")
    skill = loader.select(issue)
    assert skill is not None
    assert skill.name == "code-review"
    assert "Focus on security" in skill.body

No match field.""")
