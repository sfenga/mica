### 状态机(实现依据)
```
backlog   → {todo}
todo      → {in_progress}
in_progress → {in_review, done, blocked}
in_review → {done, in_progress, blocked}
blocked   → {todo, cancelled}
done      → {}  (终态)
cancelled → {}  (终态)
```
`done`/`cancelled` 终态;issue 默认 `todo`(创建即可派活)。

### 完整代码 `sessions/zh/s01_issue_board.py`

```python
"""s01 — Issue 板子卡 + 7 态状态机 + 多态 assignee + 看板

对应 multiCa:issue lifecycle / agents as teammates。
状态机:backlog→todo→in_progress→in_review|done|blocked;in_review→done|in_progress|blocked;
blocked→todo|cancelled;done/cancelled 终态。
多态 assignee:assignee_type ∈ {"", "member", "agent"} + assignee_id。
本章 store 为内存版;s04 起换 JSONL 后端。
"""
import sys
import uuid
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


class IssueStatus(str, Enum):
    BACKLOG = "backlog"
    TODO = "todo"
    IN_PROGRESS = "in_progress"
    IN_REVIEW = "in_review"
    DONE = "done"
    BLOCKED = "blocked"
    CANCELLED = "cancelled"


# 合法状态转移:键=当前状态,值=可转到的状态集合。其它转移一律拒绝。
VALID_TRANSITIONS = {
    IssueStatus.BACKLOG: {IssueStatus.TODO},
    IssueStatus.TODO: {IssueStatus.IN_PROGRESS},
    IssueStatus.IN_PROGRESS: {IssueStatus.IN_REVIEW, IssueStatus.DONE, IssueStatus.BLOCKED},
    IssueStatus.IN_REVIEW: {IssueStatus.DONE, IssueStatus.IN_PROGRESS, IssueStatus.BLOCKED},
    IssueStatus.BLOCKED: {IssueStatus.TODO, IssueStatus.CANCELLED},
    IssueStatus.DONE: set(),
    IssueStatus.CANCELLED: set(),
}


@dataclass
class Workspace:
    id: str
    slug: str
    name: str


@dataclass
class Agent:
    id: str
    name: str
    cli: str = "claude"
    runtime_id: str = "local"


@dataclass
class Member:
    id: str
    name: str


@dataclass
class Issue:
    id: str
    workspace_id: str
    title: str
    body: str = ""
    status: IssueStatus = IssueStatus.TODO
    assignee_type: str = ""       # "" | "member" | "agent"
    assignee_id: str = ""
    created_at: str = field(default_factory=now_iso)
    updated_at: str = field(default_factory=now_iso)
    error: str = ""


class IllegalTransition(Exception):
    pass


class IssueStore:
    """内存 issue 存储 + 状态机校验。s04 加 JSONL 后端。"""

    def __init__(self) -> None:
        self._issues: dict[str, Issue] = {}

    def create(self, workspace_id: str, title: str, body: str = "",
               assignee_type: str = "", assignee_id: str = "") -> Issue:
        issue = Issue(
            id=str(uuid.uuid4()),
            workspace_id=workspace_id,
            title=title,
            body=body,
            assignee_type=assignee_type,
            assignee_id=assignee_id,
        )
        self._issues[issue.id] = issue
        return issue

    def list(self, workspace_id: str | None = None) -> list[Issue]:
        return [
            i for i in self._issues.values()
            if workspace_id is None or i.workspace_id == workspace_id
        ]

    def get(self, issue_id: str) -> Issue | None:
        return self._issues.get(issue_id)

    def assign(self, issue_id: str, assignee_type: str, assignee_id: str) -> Issue:
        issue = self._require(issue_id)
        issue.assignee_type = assignee_type
        issue.assignee_id = assignee_id
        issue.updated_at = now_iso()
        return issue

    def transition(self, issue_id: str, new_status: IssueStatus) -> Issue:
        issue = self._require(issue_id)
        if new_status not in VALID_TRANSITIONS[issue.status]:
            raise IllegalTransition(
                f"非法转移:{issue.status.value} -> {new_status.value}"
            )
        issue.status = new_status
        issue.updated_at = now_iso()
        return issue

    def _require(self, issue_id: str) -> Issue:
        issue = self._issues.get(issue_id)
        if issue is None:
            raise KeyError(f"issue 不存在:{issue_id}")
        return issue


def render_board(issues: list[Issue]) -> str:
    """终端看板:一行一个 issue。"""
    if not issues:
        return "(空看板)"
    lines = ["ID            STATUS       ASSIGNEE                TITLE"]
    for i in issues:
        assignee = f"{i.assignee_type}:{i.assignee_id}" if i.assignee_type else "-"
        lines.append(f"{i.id[:12]:<13} {i.status.value:<12} {assignee:<23} {i.title}")
    return "\n".join(lines)


# --- CLI(单进程内执行;s04 持久化后可跨进程)---
def cmd_main(argv: list[str]) -> int:
    store = IssueStore()
    ws_id = "ws-1"
    if not argv:
        # 无参:演示场景——一条 issue 走 todo→in_progress→in_review→done
        a = store.create(ws_id, "修登录 bug", assignee_type="agent", assignee_id="agent-1")
        store.transition(a.id, IssueStatus.IN_PROGRESS)
        store.transition(a.id, IssueStatus.IN_REVIEW)
        store.transition(a.id, IssueStatus.DONE)
        store.create(ws_id, "写文档", assignee_type="member", assignee_id="alice")
        print(render_board(store.list()))
        return 0
    cmd = argv[0]
    if cmd == "create":
        title = argv[1] if len(argv) > 1 else "(untitled)"
        at = argv[2] if len(argv) > 2 else ""
        aid = argv[3] if len(argv) > 3 else ""
        issue = store.create("ws-1", title, assignee_type=at, assignee_id=aid)
        print(f"created {issue.id} '{issue.title}'")
        return 0
    if cmd == "list":
        print(render_board(store.list()))
        return 0
    if cmd == "assign":
        store.assign(argv[1], argv[2], argv[3])
        print("assigned")
        return 0
    if cmd == "show":
        issue = store.get(argv[1])
        if issue is None:
            print(f"not found: {argv[1]}", file=sys.stderr)
            return 1
        print(f"{issue.id}  {issue.status.value}  "
              f"{issue.assignee_type}:{issue.assignee_id}  {issue.title}")
        return 0
    print(f"未知命令:{cmd}", file=sys.stderr)
    return 2


if __name__ == "__main__":
    sys.exit(cmd_main(sys.argv[1:]))
```

import pytest
from sessions.zh.s01_issue_board import (
    IssueStore, IssueStatus, IllegalTransition,
)


def test_create_defaults_todo():
    store = IssueStore()
    issue = store.create("ws-1", "修登录 bug")
    assert issue.status == IssueStatus.TODO
    assert issue.assignee_type == ""


def test_backlog_to_todo():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    issue.status = IssueStatus.BACKLOG  # 直接置 backlog 起点测转移
    store.transition(issue.id, IssueStatus.TODO)
    assert store.get(issue.id).status == IssueStatus.TODO


def test_todo_to_in_progress():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    assert store.get(issue.id).status == IssueStatus.IN_PROGRESS


def test_in_progress_to_in_review():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    store.transition(issue.id, IssueStatus.IN_REVIEW)
    assert store.get(issue.id).status == IssueStatus.IN_REVIEW


def test_review_approved_to_done():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    store.transition(issue.id, IssueStatus.IN_REVIEW)
    store.transition(issue.id, IssueStatus.DONE)
    assert store.get(issue.id).status == IssueStatus.DONE


def test_review_rework_back_to_in_progress():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    store.transition(issue.id, IssueStatus.IN_REVIEW)
    store.transition(issue.id, IssueStatus.IN_PROGRESS)   # 要求返工
    assert store.get(issue.id).status == IssueStatus.IN_PROGRESS


def test_in_progress_to_blocked_and_unblock():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    store.transition(issue.id, IssueStatus.BLOCKED)
    store.transition(issue.id, IssueStatus.TODO)          # 解除阻塞,重新入队
    assert store.get(issue.id).status == IssueStatus.TODO


def test_blocked_to_cancelled():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    store.transition(issue.id, IssueStatus.BLOCKED)
    store.transition(issue.id, IssueStatus.CANCELLED)
    assert store.get(issue.id).status == IssueStatus.CANCELLED


def test_illegal_done_to_todo_rejected():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.transition(issue.id, IssueStatus.IN_PROGRESS)
    store.transition(issue.id, IssueStatus.DONE)
    with pytest.raises(IllegalTransition):
        store.transition(issue.id, IssueStatus.TODO)


def test_illegal_skip_to_done_rejected():
    # todo 不能直接 done(必须经过 in_progress)
    store = IssueStore()
    issue = store.create("ws-1", "t")
    with pytest.raises(IllegalTransition):
        store.transition(issue.id, IssueStatus.DONE)


def test_illegal_backlog_to_in_progress_rejected():
    # backlog 必须先到 todo
    store = IssueStore()
    issue = store.create("ws-1", "t")
    issue.status = IssueStatus.BACKLOG
    with pytest.raises(IllegalTransition):
        store.transition(issue.id, IssueStatus.IN_PROGRESS)


def test_assign_to_agent_sets_polymorphic_fields():
    store = IssueStore()
    issue = store.create("ws-1", "t")
    store.assign(issue.id, "agent", "agent-1")
    got = store.get(issue.id)
    assert got.assignee_type == "agent"
    assert got.assignee_id == "agent-1"


def test_list_filters_by_workspace():
    store = IssueStore()
    store.create("ws-1", "a")
    store.create("ws-2", "b")
    assert len(store.list("ws-1")) == 1
    assert len(store.list()) == 2
